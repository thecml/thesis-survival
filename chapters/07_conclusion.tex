\begin{frame}{Answers to the research questions}
\small
\begin{itemize}
  \item \textbf{RQ1 -- Uncertainty}
  \begin{itemize}
    \item MC dropout and SNGP are cheaper alternatives to VI: same C-index on MIMIC-IV with half the parameters; SNGP D-calibrated on 4/4 datasets.
    \item Conformal coverage holds on average, not per subgroup.
  \end{itemize}
  \item \textbf{RQ2 -- Multiple events}
  \begin{itemize}
    \item MENSA: one network, Weibull mixtures, trajectory loss -- competitive at lower cost.
    \item ALS: five curves per patient, and what-if analyses.
  \end{itemize}
  \item \textbf{RQ3 -- Evaluation}
  \begin{itemize}
    \item Match the metric to the objective -- about 72\% of 92 papers do not.
    \item IBS-Dep: 12--16\% less estimation error under dependent censoring, but not better at ranking models.
  \end{itemize}
  \item \textbf{RQ4 -- Beyond mortality}
  \begin{itemize}
    \item Falls (C-index 0.64 at 95\% censoring) and bearings (0.70 / 0.76).
    \item The conclusions depend on how well the endpoint represents the real outcome.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Future work}
\begin{itemize}
  \item \textbf{Interpretability}: interpretable deep survival models vs.\ post-hoc explanations (SurvSHAP \cite{Krzyzinski2023SurvSHAPt}, SurvLIME \cite{Kovalev2020SurvLIME}) -- and how to evaluate them.
  \vspace{0.6em}
  \item \textbf{Evaluation}: evaluate event times and probabilities directly; dependence-aware C-index and D-calibration by swapping the KM correction for a copula-based one.
  \vspace{0.6em}
  \item \textbf{Foundation models}: pretrained (MOTOR \cite{steinberg2024motor}) and in-context survival models (SurvivalPFN \cite{qi2026survivalpfn}, SurvPFN \cite{bohm2026survpfn}) -- do they generalize across censoring mechanisms?
\end{itemize}
\end{frame}
