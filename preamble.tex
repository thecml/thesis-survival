% ------ ROADMAP (four research questions) ------
% \roadmap[scale]{n}: n = 0 (all equal), 1..4 (highlight RQ n), 5 (all answered)
\newcommand{\rqbox}[6]{% #1 id, #2 active?, #3 title, #4 chapter, #5 papers, #6 position
  \ifnum#2=1
    \node[rqactive] (rq#1) at #6 {\textbf{RQ#1 -- #3}\hfill{\scriptsize #4}\\[0.25em]{\scriptsize #5}};
  \else
    \node[rqidle] (rq#1) at #6 {\textbf{RQ#1 -- #3}\hfill{\scriptsize #4}\\[0.25em]{\scriptsize #5}};
  \fi}
\tikzset{
  rqbase/.style={draw, rounded corners=4pt, text width=5.15cm, minimum height=2.0cm,
                 align=left, inner sep=5pt, font=\small, line width=0.8pt},
  rqactive/.style={rqbase, draw=structure.fg, fill=structure.fg, text=white},
  rqidle/.style={rqbase, draw=structure.fg!35, fill=structure.fg!6, text=black!75},
}
